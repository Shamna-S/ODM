function getInventory() {
  const props = PropertiesService.getScriptProperties();
  const token = props.getProperty('NOTION_TOKEN').trim();
  const databaseId = props.getProperty('NOTION_DATABASE_ID').trim();

  const url = 'https://api.notion.com/v1/databases/' +
              databaseId + '/query';

  const response = UrlFetchApp.fetch(url, {
    method: 'post',
    headers: {
      'Authorization': 'Bearer ' + token,
      'Notion-Version': '2022-06-28',
      'Content-Type': 'application/json'
    },
    payload: JSON.stringify({ page_size: 100 }),
    muteHttpExceptions: true
  });

  const status = response.getResponseCode();
  const data = JSON.parse(response.getContentText());

  if (status !== 200) {
    throw new Error('Inventory fetch failed: ' + status +
                    ' - ' + (data.message || 'Unknown error'));
  }

  return data.results.map(function(page) {
    const p = page.properties;

    const nameProperty = p.Name;
    const name = nameProperty &&
      nameProperty.title &&
      nameProperty.title.length
        ? nameProperty.title.map(function(item) {
            return item.plain_text;
          }).join('')
        : 'Unnamed product';

    return {
      id: page.id,
      name: name,
      price: p.Price && p.Price.number != null
        ? p.Price.number : 0,
      quantity: p.Quantity && p.Quantity.number != null
        ? p.Quantity.number : 0,
      forResale: p['For resale']
        ? p['For resale'].checkbox : false
    };
  });
}
