
<div class="section">
  <h3>Inventory</h3>

  <input
    type="text"
    id="search"
    placeholder="Search Hot Wheels..."
    oninput="filterInventory()"
    style="width:100%;padding:12px;margin-bottom:16px;border:1px solid #ddd;border-radius:8px;"
  >

  <div id="inventoryList">
    Loading inventory...
  </div>
</div>

<script>
  let inventory = [];

  function loadInventory() {
    google.script.run
      .withSuccessHandler(function(products) {
        inventory = products;

        document.getElementById('products').textContent =
          products.length;

        document.getElementById('stock').textContent =
          products.reduce((sum, p) => sum + p.quantity, 0);

        displayInventory(products);
      })
      .withFailureHandler(function(error) {
        document.getElementById('inventoryList').textContent =
          'Could not load inventory: ' + error.message;
      })
      .getInventory();
  }

  function displayInventory(products) {
    const container = document.getElementById('inventoryList');

    if (!products.length) {
      container.textContent = 'No products found.';
      return;
    }

    container.innerHTML = products.map(function(p) {
      const item = document.createElement('div');
      item.style.cssText =
        'padding:14px 0;border-bottom:1px solid #eee;';

      const name = document.createElement('strong');
      name.textContent = p.name;

      const details = document.createElement('div');
      details.style.color = '#687386';
      details.style.marginTop = '6px';
      details.textContent =
        'Price: ₹' + p.price +
        ' | Stock: ' + p.quantity +
        ' | ' + (p.forResale ? 'For resale' : 'Personal collection');

      item.appendChild(name);
      item.appendChild(details);

      return item.outerHTML;
    }).join('');
  }

  function filterInventory() {
    const search = document.getElementById('search')
      .value.toLowerCase();

    const filtered = inventory.filter(function(p) {
      return p.name.toLowerCase().includes(search);
    });

    displayInventory(filtered);
  }

  window.addEventListener('load', loadInventory);
</script>
